    <script>
        // Initialize Lucide Icons
        lucide.createIcons();


        // --- Wizard State ---
        let currentStep = 1;
        const totalSteps = 3;
        
        const form = document.getElementById('onboarding-form');
        const nextBtn = document.getElementById('next-btn');
        const backBtn = document.getElementById('back-btn');
        const progressSteps = document.querySelectorAll('#progress-steps .step');


        const stepSections = {
            1: document.querySelector('section[data-step="1"]'),
            2: document.querySelector('section[data-step="2"]'),
            3: document.querySelector('section[data-step="3"]')
        };


        const buttonText = {
            1: 'Continue to Styling',
            2: 'Continue to Payments',
            3: 'Finish & Connect Stripe'
        };


        // --- Wizard Functions ---
        
        function showStep(stepNumber) {
            currentStep = stepNumber;
            
            // Hide all sections
            Object.values(stepSections).forEach(section => section.classList.add('hidden'));
            
            // Show current section
            stepSections[currentStep].classList.remove('hidden');
            
            // Update progress bar
            progressSteps.forEach((step, index) => {
                if (index < currentStep) {
                    step.classList.add('step-primary');
                } else {
                    step.classList.remove('step-primary');
                }
            });
            
            // Update button text
            nextBtn.innerHTML = `${buttonText[currentStep]} <i data-lucide="arrow-right" class="w-5 h-5"></i>`;
            
            // Handle Stripe button case
            if (currentStep === 3) {
                nextBtn.innerHTML = `Finish & Connect Stripe <i data-lucide="lock" class="w-5 h-5"></i>`;
            }


            // Update icon
            lucide.createIcons();
            
            // Show/hide back button
            backBtn.classList.toggle('hidden', currentStep === 1);
        }


        function validateStep(stepNumber) {
            if (stepNumber === 1) {
                const brandName = document.getElementById('brand_name').value;
                const username = document.getElementById('username').value;
                
                if (!brandName || !username) {
                    alert('Please fill out your Brand Name and Username.');
                    return false;
                }
                if (username.length < 3) {
                    alert('Username must be at least 3 characters.');
                    return false;
                }
            }
            // Step 2 has defaults, so no hard validation needed
            return true;
        }


        function goToNextStep() {
            if (!validateStep(currentStep)) {
                return;
            }


            if (currentStep < totalSteps) {
                showStep(currentStep + 1);
            } else {
                // Final step action
                console.log('Onboarding complete. Redirecting to Stripe...');
                nextBtn.innerHTML = 'Redirecting... <span class="loading loading-spinner"></span>';
                
                // --- !! NOTE: THIS MUST BE A FULL PAGE REDIRECT !! ---
                // In a real app, this MUST be a full page redirect to the 
                // Stripe Connect OAuth URL, not a fetch().
                // For the demo, we'll just go to the dashboard.
                setTimeout(() => {
                    window.location.href = "app-dashboard-empty.html"; // Simulate redirect to dashboard
                }, 1500);
            }
        }


        function goToPrevStep() {
            if (currentStep > 1) {
                showStep(currentStep - 1);
            }
        }


        // --- Event Listeners ---
        
        nextBtn.addEventListener('click', goToNextStep);
        backBtn.addEventListener('click', goToPrevStep);


        // --- Step 1: Username Check ---
        const usernameInput = document.getElementById('username');
        const usernameStatus = document.getElementById('username-status');
        let typingTimer;


        usernameInput.addEventListener('keyup', () => {
            clearTimeout(typingTimer);
            usernameStatus.textContent = 'Checking availability...';
            usernameStatus.classList.remove('text-green-400', 'text-red-400');
            usernameStatus.classList.add('text-slate-500');


            typingTimer = setTimeout(() => {
                const username = usernameInput.value;
                if (username.length < 3) {
                    usernameStatus.textContent = 'Must be at least 3 characters.';
                    usernameStatus.classList.add('text-red-400');
                } else if (username === 'admin' || username === 'test') {
                    usernameStatus.textContent = `Oof! "${username}" is already taken.`;
                    usernameStatus.classList.add('text-red-400');
                } else {
                    usernameStatus.textContent = `Sweet! pagefairy.me/${username} is available!`;
                    usernameStatus.classList.add('text-green-400');
                }
            }, 800);
        });


        // --- Step 1: File Upload ---
        const dropzone = document.querySelector('.file-dropzone');
        const fileInput = document.getElementById('logo_upload');
        
        dropzone.addEventListener('click', () => fileInput.click());


        fileInput.addEventListener('change', () => {
            if (fileInput.files && fileInput.files[0]) {
                const fileName = fileInput.files[0].name;
                dropzone.querySelector('span.font-semibold').textContent = fileName;
                dropzone.querySelector('span.text-sm').textContent = 'Click to change file';
                dropzone.querySelector('i').setAttribute('data-lucide', 'file-check-2');
                lucide.createIcons();
            }
        });


        // --- Step 2: Color Picker ---
        const colorPicker = document.getElementById('brand_color_picker');
        const colorHex = document.getElementById('brand_color_hex');


        colorPicker.addEventListener('input', (e) => {
            colorHex.value = e.target.value.toUpperCase();
        });


        colorHex.addEventListener('input', (e) => {
            // Basic hex validation
            if (/^#[0-9A-F]{6}$/i.test(e.target.value) || /^#[0-9A-F]{3}$/i.test(e.target.value)) {
                colorPicker.value = e.target.value;
            }
        });


        // --- Step 2: Font Slider ---
        const fontSlider = document.getElementById('font-slider');
        const fontDots = document.querySelectorAll('#font-dots span');
        const fontCards = document.querySelectorAll('.font-card');
        const fontInput = document.getElementById('font_style');
        
        let scrollTimer;
        fontSlider.addEventListener('scroll', () => {
            clearTimeout(scrollTimer);
            scrollTimer = setTimeout(() => {
                const scrollLeft = fontSlider.scrollLeft;
                const slideWidth = fontSlider.clientWidth;
                const currentIndex = Math.round(scrollLeft / slideWidth);


                // Update dots
                fontDots.forEach((dot, index) => {
                    dot.classList.toggle('bg-purple-600', index === currentIndex);
                    dot.classList.toggle('bg-slate-700', index !== currentIndex);
                });


                // Update selected card
                fontCards.forEach((card, index) => {
                    card.classList.toggle('selected', index === currentIndex);
                    if (index === currentIndex) {
                        fontInput.value = card.dataset.font;
                    }
                });
            }, 100); // Debounce scroll event
        });
        
        // Add click to dots
        fontDots.forEach(dot => {
            dot.addEventListener('click', () => {
                const slideIndex = parseInt(dot.dataset.slideIndex, 10);
                const slideWidth = fontSlider.clientWidth;
                fontSlider.scrollTo({
                    left: slideIndex * slideWidth,
                    behavior: 'smooth'
                });
            });
        });


        // --- Initial Call ---
        showStep(1);
    </script>
